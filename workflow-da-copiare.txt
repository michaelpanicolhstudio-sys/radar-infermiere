name: Radar giornaliero

on:
  schedule:
    # GitHub usa l'ora UTC. Due orari per coprire ora solare e ora legale:
    # il programma parte solo al primo passaggio dopo le 7:00 di Roma.
    - cron: "7 5 * * *"
    - cron: "7 6 * * *"
  workflow_dispatch: {}   # avvio a mano dal pulsante "Run workflow" (manda anche l'email)
  push:                   # ogni modifica a programma o configurazione fa partire un giro di prova, senza email
    branches: [main]
    paths:
      - "radar/**"
      - "webapp/**"
      - "config.yaml"
      - "data/comuni.csv"
      - "data/strutture.csv"
      - "requirements.txt"
      - ".github/workflows/radar.yml"

permissions:
  contents: write

concurrency:
  group: radar
  cancel-in-progress: false

jobs:
  radar:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    env:
      GMAIL_USER: ${{ secrets.GMAIL_USER }}
      GMAIL_APP_PASSWORD: ${{ secrets.GMAIL_APP_PASSWORD }}
      PEC_USER: ${{ secrets.PEC_USER }}
      PEC_PASSWORD: ${{ secrets.PEC_PASSWORD }}
      PEC_IMAP_HOST: ${{ secrets.PEC_IMAP_HOST }}
      EMAIL_TO: ${{ secrets.EMAIL_TO }}
      APP_URL: ${{ secrets.APP_URL }}
      CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
      CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
      PROGETTO_PAGES: ${{ vars.PROGETTO_PAGES || 'radar-infermiere' }}
    steps:
      - name: Scarica il repository
        uses: actions/checkout@v6

      - name: Prepara Python
        uses: actions/setup-python@v6
        with:
          python-version: "3.12"
          cache: pip

      - name: Installa le librerie
        run: pip install -r requirements.txt

      - name: Prove rapide (senza rete)
        run: python tests/prova.py

      - name: Giro del Radar
        id: giro
        run: python -m radar ${{ github.event_name == 'schedule' && '--pianificato' || '' }} ${{ github.event_name == 'push' && '--senza-email' || '' }}

      - name: Salva la memoria nel repository
        if: steps.giro.outputs.eseguito == 'true'
        run: |
          git config user.name "Radar Infermiere"
          git config user.email "radar@users.noreply.github.com"
          git add data
          git diff --cached --quiet || git commit -m "Radar: giro del $(TZ=Europe/Rome date +%d/%m/%Y)"
          git pull --rebase origin main || true
          git push

      - name: Controlla se il progetto Cloudflare Pages esiste già
        id: pages
        if: steps.giro.outputs.eseguito == 'true' && env.CLOUDFLARE_API_TOKEN != ''
        run: |
          codice=$(curl -s -o /dev/null -w "%{http_code}" -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
            "https://api.cloudflare.com/client/v4/accounts/$CLOUDFLARE_ACCOUNT_ID/pages/projects/$PROGETTO_PAGES" || echo 000)
          echo "Risposta di Cloudflare: $codice"
          echo "esiste=$([ "$codice" = "200" ] && echo true || echo false)" >> "$GITHUB_OUTPUT"

      - name: Crea il progetto Cloudflare Pages (solo la prima volta)
        if: steps.giro.outputs.eseguito == 'true' && env.CLOUDFLARE_API_TOKEN != '' && steps.pages.outputs.esiste != 'true'
        continue-on-error: true
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          command: pages project create ${{ env.PROGETTO_PAGES }} --production-branch=main

      - name: Pubblica la web app su Cloudflare Pages
        if: steps.giro.outputs.eseguito == 'true' && env.CLOUDFLARE_API_TOKEN != ''
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          command: pages deploy dist --project-name=${{ env.PROGETTO_PAGES }} --branch=main --commit-dirty=true
